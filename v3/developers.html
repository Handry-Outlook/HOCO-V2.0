Put HOCO thunderstorm outlooks in your own app

Every AUTO-HOCO run is published as a GeoJSON file: shaded areas for the chance of lightning within 20 km, for today and the next two days. Find the runs in Firestore, download the file from Firebase Storage, and draw it with any map library.

Try it liveSee the code

How it works

1Find runsQuery the hoco_requests collection for public, completed AUTO-HOCO runs.
2Get the fileBuild the Storage URL from the run ID, read its download token, fetch the GeoJSON.
3Draw itAdd the GeoJSON to Leaflet, Mapbox or MapLibre and colour it by risk.

Quick start

Seven steps from nothing to a map. Each one has working code in Code examples.

  1. Connect to the HOCO Firebase projectUse the Firebase Web SDK with the config below, or the Firestore REST API from any language.
  2. Query hoco_requestsEvery forecast run is one document in this collection.
  3. Keep only public, completed AUTO-HOCO runs
    Filters
    public == true
status == "completed"
type   == "automated"
  4. Read the document IDThe document ID is the run ID, for example AUTO-1782803122360-England-Day1-2. See Run IDs.
  5. Build the Storage metadata URL
    URL
    https://firebasestorage.googleapis.com/v0/b/hoco-3b23e.firebasestorage.app/o/forecasts%2F{runId}.geojson
  6. Fetch the metadata and read downloadTokensAdd ?alt=media&token={token} to the same URL to get the file itself.
  7. Fetch the GeoJSON and draw itColour each area by its risk value, or use the fill colour that comes with it.
💡Don't guess run IDsThe number in a run ID is not predictable. Always discover runs from Firestore first.

Data format

Each file is a GeoJSON FeatureCollection. Newer runs use the banded layout: one polygon per chance band, nested from low to high.

Run document fields you can rely on

FieldExampleMeaning
regionEnglandRegion ID
startTime, endTime2026-11-18T06:00The period the forecast covers
createdAt, completedAtTimestampWhen the run started and finished
geojsonUrlhttps://…A direct link to the file (you can use this instead of steps 5 and 6)
geojsonLayoutbands_v2Which layout the file uses

Feature properties

PropertyExampleMeaning
feature_typepolygonpolygon for shaded areas, point for the grid values underneath
risk30Chance of lightning within 20 km, in %. On polygons it is the band's lower edge
band_min, band_max, band_label30, 40, "30-40%"The band a polygon belongs to
fill, stroke, fill-opacity#ffb327HOCO's own colours, ready to use

Draw only feature_type == "polygon" for the shaded map. Grid points are many thousands of features and are meant for looking up a value at one spot.

function parseAutoHocoRunId(runId) {
  const match = String(runId).match(/^AUTO-(\d+)-(.+)-(Day\d+-\d+)$/);
  return {
    autoCode: match?.[1] || null,
    region: match?.[2] || null,
    forecastDay: match?.[3] || null
  };
}

function buildMetadataUrl(runId) {
  const bucket = "hoco-3b23e.firebasestorage.app";
  const objectPath = `forecasts/${runId}.geojson`;
  return `https://firebasestorage.googleapis.com/v0/b/${bucket}/o/${encodeURIComponent(objectPath)}`;
}

async function fetchAutoHocoRuns() {
  const q = query(
    collection(db, "hoco_requests"),
    where("public", "==", true),
    where("status", "==", "completed"),
    where("type", "==", "automated"),
    orderBy("completedAt", "desc"),
    limit(20)
  );

  const snapshot = await getDocs(q);

  return snapshot.docs.map((docSnap) => {
    const runId = docSnap.id;
    const parsed = parseAutoHocoRunId(runId);

    const data = docSnap.data();

    return {
      runId,
      ...parsed,
      region: data.region,
      startTime: data.startTime,
      endTime: data.endTime,
      completedAt: data.completedAt,
      geojsonUrl: data.geojsonUrl,
      metadataUrl: buildMetadataUrl(runId)
    };
  });
}

const runs = await fetchAutoHocoRuns();
console.log(runs);
</script>

def parse_auto_hoco_run_id(run_id):
    match = re.match(r"^AUTO-(\d+)-(.+)-(Day\d+-\d+)$", run_id)

    if not match:
        return {
            "autoCode": None,
            "regionFromId": None,
            "forecastDay": None
        }

    return {
        "autoCode": match.group(1),
        "regionFromId": match.group(2),
        "forecastDay": match.group(3)
    }

def firestore_value_to_python(value):
    if "stringValue" in value:
        return value["stringValue"]

    if "booleanValue" in value:
        return value["booleanValue"]

    if "integerValue" in value:
        return int(value["integerValue"])

    if "doubleValue" in value:
        return float(value["doubleValue"])

    if "timestampValue" in value:
        return value["timestampValue"]

    if "arrayValue" in value:
        return [
            firestore_value_to_python(item)
            for item in value.get("arrayValue", {}).get("values", [])
        ]

    if "mapValue" in value:
        return {
            key: firestore_value_to_python(item)
            for key, item in value.get("mapValue", {}).get("fields", {}).items()
        }

    return None

def firestore_doc_to_python(document):
    fields = document.get("fields", {})

    data = {
        key: firestore_value_to_python(value)
        for key, value in fields.items()
    }

    document_path = document.get("name", "")
    run_id = document_path.split("/")[-1]

    data["runId"] = data.get("id") or run_id

    return data

def build_storage_metadata_url(run_id):
    object_path = f"forecasts/{run_id}.geojson"
    encoded_path = quote(object_path, safe="")
    return f"https://firebasestorage.googleapis.com/v0/b/{STORAGE_BUCKET}/o/{encoded_path}"

def build_geojson_download_url_from_metadata(metadata):
    bucket = metadata["bucket"]
    name = metadata["name"]

    token = metadata.get("downloadTokens", "").split(",")[0]
    encoded_name = quote(name, safe="")

    return f"https://firebasestorage.googleapis.com/v0/b/{bucket}/o/{encoded_name}?alt=media&token={token}"

def fetch_storage_metadata(metadata_url):
    response = requests.get(metadata_url)
    response.raise_for_status()
    return response.json()

def fetch_geojson(download_url):
    response = requests.get(download_url)
    response.raise_for_status()
    return response.json()

def fetch_auto_hoco_catalogue(region=None, forecast_day=None, limit_count=20):
    url = (
        f"https://firestore.googleapis.com/v1/projects/{PROJECT_ID}"
        f"/databases/{DATABASE_ID}/documents:runQuery?key={API_KEY}"
    )

    filters = [
        {
            "fieldFilter": {
                "field": { "fieldPath": "public" },
                "op": "EQUAL",
                "value": { "booleanValue": True }
            }
        },
        {
            "fieldFilter": {
                "field": { "fieldPath": "status" },
                "op": "EQUAL",
                "value": { "stringValue": "completed" }
            }
        },
        {
            "fieldFilter": {
                "field": { "fieldPath": "type" },
                "op": "EQUAL",
                "value": { "stringValue": "automated" }
            }
        }
    ]

    if region:
        filters.append({
            "fieldFilter": {
                "field": { "fieldPath": "region" },
                "op": "EQUAL",
                "value": { "stringValue": region }
            }
        })

    body = {
        "structuredQuery": {
            "from": [
                { "collectionId": "hoco_requests" }
            ],
            "where": {
                "compositeFilter": {
                    "op": "AND",
                    "filters": filters
                }
            },
            "orderBy": [
                {
                    "field": { "fieldPath": "completedAt" },
                    "direction": "DESCENDING"
                }
            ],
            "limit": limit_count
        }
    }

    response = requests.post(url, json=body)
    response.raise_for_status()

    raw_results = response.json()
    runs = []

    for item in raw_results:
        document = item.get("document")

        if not document:
            continue

        run = firestore_doc_to_python(document)
        run_id = run["runId"]
        parsed = parse_auto_hoco_run_id(run_id)

        catalogue_item = {
            "runId": run_id,
            "autoCode": parsed["autoCode"],
            "region": run.get("region") or parsed["regionFromId"],
            "forecastDay": parsed["forecastDay"],
            "weather": run.get("weather"),
            "startTime": run.get("startTime"),
            "endTime": run.get("endTime"),
            "createdAt": run.get("createdAt"),
            "completedAt": run.get("completedAt"),
            "metadataUrl": build_storage_metadata_url(run_id),
            "existingGeojsonUrl": run.get("geojsonUrl")
        }

        if forecast_day and catalogue_item["forecastDay"] != forecast_day:
            continue

        runs.append(catalogue_item)

    return runs

if __name__ == "__main__":
    runs = fetch_auto_hoco_catalogue(

JavaScript · draw a run with Leaflet
// HOCO's colour bands (chance of lightning within 20 km, %)
const HOCO_BANDS = [
  [5, "#9fd8c4"], [10, "#66c2a4"], [15, "#b3e05a"], [20, "#fff200"], [30, "#ffb327"], [40, "#ff7f27"],
  [50, "#ff4127"], [60, "#e0182d"], [70, "#b0186b"], [80, "#8a2bbd"], [90, "#4b1fd1"]
];

function getRiskColor(risk) {
  let color = "#b8c2cc";
  for (const [min, fill] of HOCO_BANDS) if (risk >= min) color = fill;
  return color;
}

// Draw only the shaded areas, not the grid points
const areas = { ...geojson, features: geojson.features.filter((f) => f.properties?.feature_type !== "point") };

const layer = L.geoJSON(areas, {
  style: function(feature) {
    const risk = Number(
      feature.properties?.riskValue ??
      feature.properties?.peak_risk ??
      feature.properties?.risk ??
      0
    );

    return {
      color: getRiskColor(risk),
      weight: 2,
      fillColor: getRiskColor(risk),
      fillOpacity: 0.35
    };
  },
  onEachFeature: function(feature, layer) {
    layer.bindPopup(JSON.stringify(feature.properties, null, 2));
  }
}).addTo(map);

map.fitBounds(layer.getBounds());
