Put HOCO thunderstorm & severe outlooks in your own app

Every scheduled HOCO run publishes GeoJSON outlooks for the chance of lightning within 20 km across today and the next two days — covering Traditional AUTO-HOCO, AI HOCO (Beta), Hourcast (Beta) sub-daily windows, and ESTOFEX-style Severity (Level 1–3). Discover runs in Firestore, download GeoJSON over HTTPS, and render with any map library.

Try it liveSee the codeCompare products

How it works

1Find forecastsQuery hoco_requests for Traditional, AI HOCO & Severity, or read system_stats/storm_timing_* for Hourcast.
2Download GeoJSONUse the document's geojsonUrl (or downloadTokens for older archives) plus any severity.geojsonUrl.
3Draw the mapStyle feature_type == "polygon" by risk and overlay severity_polygon levels 1–3.

Quick start

Seven steps from nothing to a map for full-day Traditional or AI HOCO runs, plus Hourcast and Severity overlays.

  1. Connect to the HOCO Firebase projectUse the Firebase Web SDK with the config below, or the Firestore REST API from any language.
  2. Query hoco_requestsEvery forecast run is one document in this collection.
  3. Filter public, completed runs
    Filters
    public == true
status == "completed"
type   == "automated"   # use "ai" for AI HOCO (England & Scotland)
  4. Read the document IDFor example AUTO-1782803122360-England-Day1-2 or AI-AUTO-1782803122360-England-Day1-2. See Run IDs.
  5. Use geojsonUrl or build the Storage metadata URLRecent runs include a signed HTTPS geojsonUrl (valid 7 days). For permanent archive access on older runs, fetch the Firebase Storage metadata JSON instead:
    URL
    https://firebasestorage.googleapis.com/v0/b/hoco-3b23e.firebasestorage.app/o/forecasts%2F{runId}.geojson
  6. Read downloadTokens (when using metadata)Append ?alt=media&token={token} to the metadata URL to download the GeoJSON file.
  7. Draw GeoJSON & optional overlaysColour each feature_type == "polygon" by risk, and if doc.severity?.maxLevel >= 1, fetch doc.severity.geojsonUrl for Level 1–3 threat areas.
⏱️Hourcast sub-daily windowsRead system_stats/storm_timing_{region}_{day} (lowercase region ID + day0, day1, or day2), or system_stats/storm_timing_run_{runId} for any AUTO run where stormHour == true. Each entry in windows[] already includes a persistent HTTPS geojsonUrl and optional per-window severity.
💡Don't guess run IDsThe numeric code inside a run ID is not predictable. Always discover runs from Firestore first.

function parseHocoRunId(runId) {
  const match = String(runId).match(/^(?:(AI)-)?AUTO-(\d+)-(.+)-(Day\d+-\d+)$/);

  const bucket = "hoco-3b23e.firebasestorage.app";
  const objectPath = `forecasts/${runId}.geojson`;
  return `https://firebasestorage.googleapis.com/v0/b/${bucket}/o/${encodeURIComponent(objectPath)}`;

// product: "automated" (all 6 active regions) or "ai" (England & Scotland)
async function fetchHocoRuns({ product = "automated", region = null, count = 20 } = {}) {
  const filters = [
    where("public", "==", true),
    where("status", "==", "completed"),
    where("type", "==", product)
  ];
  if (region) filters.push(where("region", "==", region));

  const q = query(collection(db, "hoco_requests"), ...filters, orderBy("completedAt", "desc"), limit(count));
  const snapshot = await getDocs(q);

  return snapshot.docs.map((docSnap) => {
    const runId = docSnap.id;
    const data = docSnap.data();
    return {
      runId,
      ...parseHocoRunId(runId),
      region: data.region,
      startTime: data.startTime,
      endTime: data.endTime,
      completedAt: data.completedAt,
      geojsonUrl: data.geojsonUrl,
      metadataUrl: buildMetadataUrl(runId),
      stormHour: Boolean(data.stormHour),
      stormHourPeak: data.stormHourPeak ?? null,
      severity: data.severity || null,
      aiRunId: data.aiRunId || null
    };
  });
}

def parse_hoco_run_id(run_id):
    match = re.match(r"^(?:(AI)-)?AUTO-(\d+)-(.+)-(Day\d+-\d+)$", run_id)
    if not match:
        return { "isAi": False, "autoCode": None, "regionFromId": None, "forecastDay": None }
    return {
        "isAi": bool(match.group(1)),
        "autoCode": match.group(2),
        "regionFromId": match.group(3),
        "forecastDay": match.group(4)
    }

def firestore_value_to_python(value):
    if "stringValue" in value: return value["stringValue"]
    if "booleanValue" in value: return value["booleanValue"]
    if "integerValue" in value: return int(value["integerValue"])
    if "doubleValue" in value: return float(value["doubleValue"])
    if "timestampValue" in value: return value["timestampValue"]
    if "arrayValue" in value:
        return [firestore_value_to_python(i) for i in value.get("arrayValue", {}).get("values", [])]
    if "mapValue" in value:
        return {k: firestore_value_to_python(v) for k, v in value.get("mapValue", {}).get("fields", {}).items()}
    return None

def firestore_doc_to_python(document):
    data = {k: firestore_value_to_python(v) for k, v in document.get("fields", {}).items()}
    run_id = document.get("name", "").split("/")[-1]
    data["runId"] = data.get("id") or run_id
    return data

def build_storage_metadata_url(object_path):
    return f"https://firebasestorage.googleapis.com/v0/b/{STORAGE_BUCKET}/o/{quote(object_path, safe='')}"

def build_download_url_from_metadata(metadata):
    token = metadata.get("downloadTokens", "").split(",")[0]
    return f"https://firebasestorage.googleapis.com/v0/b/{metadata['bucket']}/o/{quote(metadata['name'], safe='')}?alt=media&token={token}"

# run_type: "automated" (all 6 active regions) or "ai" (England & Scotland)
def fetch_hoco_catalogue(region=None, forecast_day=None, run_type="automated", limit_count=20):
    url = f"{FIRESTORE_BASE}/documents:runQuery?key={API_KEY}"
    filters = [
        { "fieldFilter": { "field": { "fieldPath": "public" }, "op": "EQUAL", "value": { "booleanValue": True } } },
        { "fieldFilter": { "field": { "fieldPath": "status" }, "op": "EQUAL", "value": { "stringValue": "completed" } } },
        { "fieldFilter": { "field": { "fieldPath": "type" }, "op": "EQUAL", "value": { "stringValue": run_type } } }
    ]
    if region:
        filters.append({ "fieldFilter": { "field": { "fieldPath": "region" }, "op": "EQUAL", "value": { "stringValue": region } } })

    body = {
        "structuredQuery": {
            "from": [{ "collectionId": "hoco_requests" }],
            "where": { "compositeFilter": { "op": "AND", "filters": filters } },
            "orderBy": [{ "field": { "fieldPath": "completedAt" }, "direction": "DESCENDING" }],
            "limit": limit_count
        }
    }
    response = requests.post(url, json=body, timeout=20)
    response.raise_for_status()

    runs = []
    for item in response.json():
        document = item.get("document")
        if not document:
            continue
        run = firestore_doc_to_python(document)
        parsed = parse_hoco_run_id(run["runId"])
        entry = {
            "runId": run["runId"],
            "type": run_type,
            "region": run.get("region") or parsed["regionFromId"],
            "forecastDay": parsed["forecastDay"],
            "startTime": run.get("startTime"),
            "endTime": run.get("endTime"),
            "stormHour": bool(run.get("stormHour")),
            "severity": run.get("severity"),
            "geojsonUrl": run.get("geojsonUrl"),
            "metadataUrl": build_storage_metadata_url(f"forecasts/{run['runId']}.geojson")
        }
        if forecast_day and entry["forecastDay"] != forecast_day:
            continue
        runs.append(entry)
    return runs

def fetch_hourcast_manifest(region="France", day="day0"):
    doc_id = f"storm_timing_{region.lower()}_{day}"

JavaScript · draw risk bands & Severity with Leaflet
// HOCO's colour bands (chance of lightning within 20 km, %)
const HOCO_BANDS = [
  [5, "#9fd8c4"], [10, "#66c2a4"], [15, "#b3e05a"], [20, "#fff200"], [30, "#ffb327"], [40, "#ff7f27"],
  [50, "#ff4127"], [60, "#e0182d"], [70, "#b0186b"], [80, "#8a2bbd"], [90, "#4b1fd1"]
];
const SEVERITY_COLOURS = { 1: "#d97706", 2: "#dc2626", 3: "#7e22ce" };

function getRiskColor(risk) {
  let color = "#b8c2cc";
  for (const [min, fill] of HOCO_BANDS) if (risk >= min) color = fill;
  return color;
}

// 1. Draw only the shaded lightning polygons, not the grid points
const areas = { ...geojson, features: geojson.features.filter((f) => f.properties?.feature_type !== "point") };
const riskLayer = L.geoJSON(areas, {
  style: (feature) => {
    const risk = Number(feature.properties?.riskValue ?? feature.properties?.peak_risk ?? feature.properties?.risk ?? 0);
    return { color: getRiskColor(risk), weight: 1.5, fillColor: getRiskColor(risk), fillOpacity: 0.45 };
  }
}).addTo(map);

// 2. Optional: overlay ESTOFEX Severity polygons when present (doc.severity?.geojsonUrl)
if (severityGeojson?.features?.length) {
  const sevPolys = {
    ...severityGeojson,
    features: severityGeojson.features.filter((f) => f.properties?.feature_type === "severity_polygon")
  };
  L.geoJSON(sevPolys, {
    style: (f) => {
      const lvl = Number(f.properties?.level || 1);
      const col = SEVERITY_COLOURS[lvl] || "#d97706";
      return { color: col, weight: lvl >= 2 ? 2.8 : 2, dashArray: lvl === 1 ? "6 4" : null, fillColor: col, fillOpacity: 0.16 };
    },
    onEachFeature: (f, l) => l.bindTooltip(`${f.properties?.label} · ${(f.properties?.hazards || []).join(", ")}`)
  }).addTo(map);
}

map.fitBounds(riskLayer.getBounds());
