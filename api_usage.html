Forecast products

Product Firestore collection & filter / doc path Active regions Cadence & notes
Traditional hoco_requests · public == true, status == "completed", type == "automated" England, Scotland, Germany, Poland, Baltics, France Every 3h (England, Scotland) or 6h (Europe) · Days 0–2 (bands_v2)
AI HOCO (Beta) hoco_requests · public == true, status == "completed", type == "ai" England, Scotland Every 3h (ID prefix AI-AUTO-...; may be skipped for individual runs)
Hourcast (Beta) system_stats/storm_timing_{region}_{day} or storm_timing_run_{runId} All 6 active regions (basis: "traditional" in Europe) 1h windows (Day 0), 2h (Day 1), 3h covering first 18h (Day 2)
ESTOFEX Severity (L1–L3) severity.geojsonUrl on hoco_requests doc or Hourcast windows[] entry All 6 active regions Populated when severity.maxLevel >= 1 (full-day also at forecasts/{runId}_severity.geojson)

Step-by-step developer workflow

1
Connect to the HOCO Firebase project using the Firebase Web SDK (or Firestore REST API).
2
Query hoco_requests for Traditional/AI runs, or read system_stats/storm_timing_{region}_{day} for Hourcast.
3
Filter public completed runs: public == true status == "completed" type == "automated" # or "ai" for AI HOCO (England & Scotland)
4
Read the Firestore document ID, for example AUTO-1782803122360-England-Day1-2 or AI-AUTO-1782803122360-England-Day1-2.
5
Parse the run ID: AUTO / AI-AUTO = automated Traditional or AI HOCO run 1782803122360 = autoCode England = region Day1-2 = forecastDay (Day0-1, Day1-2, or Day2-3)
6
Use doc.geojsonUrl (7-day signed URL) or build the permanent Firebase Storage metadata URL: https://firebasestorage.googleapis.com/v0/b/hoco-3b23e.firebasestorage.app/o/forecasts%2F{runId}.geojson
7
Read downloadTokens if using Storage metadata, download GeoJSON + optional doc.severity?.geojsonUrl, and draw polygons on your map.
External developers should not guess AUTO-HOCO numeric codes. Discover available runs from Firestore first.

JavaScript example code external developers can follow

Discovers Traditional or AI HOCO runs from hoco_requests and Hourcast windows from system_stats. 

<script type="module">
import { initializeApp } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-app.js";
import {
  getFirestore, collection, doc, getDoc, getDocs, query, where, orderBy, limit
} from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js";

const firebaseConfig = {
  apiKey: "AIzaSyA8IYTApDl8vhZENPJXIw48rR_SUhfFPVQ",
  authDomain: "hoco-3b23e.firebaseapp.com",
  projectId: "hoco-3b23e",
  storageBucket: "hoco-3b23e.firebasestorage.app",
  messagingSenderId: "409513272369",
  appId: "1:409513272369:web:a857d9b039d53710f8a110"
};

const app = initializeApp(firebaseConfig);
const db = getFirestore(app);

function parseAutoHocoRunId(runId) {
  const match = String(runId).match(/^(?:(AI)-)?AUTO-(\d+)-(.+)-(Day\d+-\d+)$/);
  return {
    isAi: Boolean(match?.[1]),
    autoCode: match?.[2] || null,
    region: match?.[3] || null,
    forecastDay: match?.[4] || null
  };
}

function buildMetadataUrl(runId) {
  const bucket = "hoco-3b23e.firebasestorage.app";
  const objectPath = `forecasts/${runId}.geojson`;
  return `https://firebasestorage.googleapis.com/v0/b/${bucket}/o/${encodeURIComponent(objectPath)}`;
}

// productType: "automated" (all 6 active regions) or "ai" (England & Scotland)
async function fetchHocoRuns(productType = "automated", region = null) {
  const filters = [
    where("public", "==", true),
    where("status", "==", "completed"),
    where("type", "==", productType)
  ];
  if (region) filters.push(where("region", "==", region));

  const q = query(collection(db, "hoco_requests"), ...filters, orderBy("completedAt", "desc"), limit(20));
  const snapshot = await getDocs(q);

  return snapshot.docs.map((docSnap) => {
    const runId = docSnap.id;
    const data = docSnap.data();
    return {
      runId,
      ...parseAutoHocoRunId(runId),
      metadataUrl: buildMetadataUrl(runId),
      severity: data.severity || null,
      stormHour: Boolean(data.stormHour),
      ...data
    };
  });
}

// Latest Hourcast manifest (e.g. France day0)
async function fetchHourcast(region = "France", day = "day0") {
  const snap = await getDoc(doc(db, "system_stats", `storm_timing_${region.toLowerCase()}_${day}`));
  return snap.exists() ? snap.data() : null;
}

console.log(await fetchHocoRuns("automated", "France"));
console.log(await fetchHourcast("England", "day0"));
</script>

def parse_auto_hoco_run_id(run_id):
    match = re.match(r"^(?:(AI)-)?AUTO-(\d+)-(.+)-(Day\d+-\d+)$", run_id)
    if not match:
        return { "isAi": False, "autoCode": None, "regionFromId": None, "forecastDay": None }
    return {
        "isAi": bool(match.group(1)),
        "autoCode": match.group(2),
        "regionFromId": match.group(3),
        "forecastDay": match.group(4)
    }

def firestore_value_to_python(value):
    if "stringValue" in value: return value["stringValue"]
    if "booleanValue" in value: return value["booleanValue"]
    if "integerValue" in value: return int(value["integerValue"])
    if "doubleValue" in value: return float(value["doubleValue"])
    if "timestampValue" in value: return value["timestampValue"]
    if "arrayValue" in value:
        return [firestore_value_to_python(item) for item in value.get("arrayValue", {}).get("values", [])]
    if "mapValue" in value:
        return {key: firestore_value_to_python(item) for key, item in value.get("mapValue", {}).get("fields", {}).items()}
    return None

def firestore_doc_to_python(document):
    fields = document.get("fields", {})
    data = {key: firestore_value_to_python(value) for key, value in fields.items()}
    run_id = document.get("name", "").split("/")[-1]
    data["runId"] = data.get("id") or run_id
    return data

def build_storage_metadata_url(run_id):
    object_path = f"forecasts/{run_id}.geojson"
    return f"https://firebasestorage.googleapis.com/v0/b/{STORAGE_BUCKET}/o/{quote(object_path, safe='')}"

def build_geojson_download_url_from_metadata(metadata):
    token = metadata.get("downloadTokens", "").split(",")[0]
    return f"https://firebasestorage.googleapis.com/v0/b/{metadata['bucket']}/o/{quote(metadata['name'], safe='')}?alt=media&token={token}"

# run_type: "automated" (all 6 active regions) or "ai" (England & Scotland)
def fetch_auto_hoco_catalogue(region=None, forecast_day=None, run_type="automated", limit_count=20):
    url = f"{FIRESTORE_BASE}/documents:runQuery?key={API_KEY}"
    filters = [
        { "fieldFilter": { "field": { "fieldPath": "public" }, "op": "EQUAL", "value": { "booleanValue": True } } },
        { "fieldFilter": { "field": { "fieldPath": "status" }, "op": "EQUAL", "value": { "stringValue": "completed" } } },
        { "fieldFilter": { "field": { "fieldPath": "type" }, "op": "EQUAL", "value": { "stringValue": run_type } } }
    ]
    if region:
        filters.append({ "fieldFilter": { "field": { "fieldPath": "region" }, "op": "EQUAL", "value": { "stringValue": region } } })

    body = {
        "structuredQuery": {
            "from": [{ "collectionId": "hoco_requests" }],
            "where": { "compositeFilter": { "op": "AND", "filters": filters } },
            "orderBy": [{ "field": { "fieldPath": "completedAt" }, "direction": "DESCENDING" }],
            "limit": limit_count
        }
    }
    response = requests.post(url, json=body, timeout=20)
    response.raise_for_status()

    runs = []
    for item in response.json():
        document = item.get("document")
        if not document:
            continue
        run = firestore_doc_to_python(document)
        parsed = parse_auto_hoco_run_id(run["runId"])
        catalogue_item = {
            "runId": run["runId"],
            "type": run_type,
            "region": run.get("region") or parsed["regionFromId"],
            "forecastDay": parsed["forecastDay"],
            "startTime": run.get("startTime"),
            "endTime": run.get("endTime"),
            "stormHour": bool(run.get("stormHour")),
            "severity": run.get("severity"),
            "metadataUrl": build_storage_metadata_url(run["runId"]),
            "existingGeojsonUrl": run.get("geojsonUrl")
        }
        if forecast_day and catalogue_item["forecastDay"] != forecast_day:
            continue
        runs.append(catalogue_item)
    return runs

def fetch_hourcast_manifest(region="France", day="day0"):
    doc_id = f"storm_timing_{region.lower()}_{day}"
    r = requests.get(f"{FIRESTORE_BASE}/documents/system_stats/{doc_id}?key={API_KEY}", timeout=20)
    if r.status_code == 404:
        return None
    r.raise_for_status()
    return firestore_doc_to_python(r.json())

Displaying the GeoJSON on a map

Filter out feature_type === "point" when rendering shaded probability bands with Leaflet, and optionally overlay severity_polygon features. 

const areas = {
  ...geojson,
  features: geojson.features.filter((f) => f.properties?.feature_type !== "point")
};

const layer = L.geoJSON(areas, {
  style: function(feature) {
    const risk = Number(
      feature.properties?.riskValue ??
      feature.properties?.peak_risk ??
      feature.properties?.risk ??
      0
    );

    return {
      color: getRiskColor(risk),
      weight: 1.5,
      fillColor: getRiskColor(risk),
      fillOpacity: 0.45
    };
  },
  onEachFeature: function(feature, layer) {
    layer.bindPopup(JSON.stringify(feature.properties, null, 2));
  }
}).addTo(map);

map.fitBounds(layer.getBounds());
